      id: item.id,
      route: result.route,
      next: result.next,
      explorationRequired: result.explorationRequired,
      anchor: item.anchor,
      prompt: drafts[item.id].prompt,

// Mechanical custody/readback checks, not a claim of semantic or behavioral validation.
const reread = JSON.parse(await readFile(new URL('output.json', directory), 'utf8'));
const protectedPhrases = {

const requiredFields = ['id', 'route', 'next', 'explorationRequired', 'anchor', 'prompt', 'clarifyingQuestions', 'notes'];
const checks = reread.cases.map((item, index) => {
  const source = input.cases[index];
  const observed = results.get(item.id);
  const request = requests.cases.find(row => row.id === item.id)?.request;
  const check = {
    id: item.id,
    exactOutputFields: Object.keys(item).length === requiredFields.length && requiredFields.every(field => Object.hasOwn(item, field)),
    correctInputOrder: item.id === source.id,
    anchorPreservedExactly: item.anchor === source.anchor && request?.context.anchor === source.context.anchor,
    observedControlsPreserved: item.route === observed.route && item.next === observed.next && item.explorationRequired === observed.explorationRequired,
    protectedPhrasePreserved: !protectedPhrases[item.id] || item.prompt.includes(protectedPhrases[item.id]),
    nonemptyPromptAndNotes: typeof item.prompt === 'string' && item.prompt.length > 0 && typeof item.notes === 'string' && item.notes.length > 0,
    noClarifyingQuestionsAdded: Array.isArray(item.clarifyingQuestions) && item.clarifyingQuestions.length === 0,
    investigationHasNoObservedEvidence: request.kind !== 'investigation' || request.state.model.observations.length === 0,
    stage: item.next === 'readback' ? 'textually-reviewed-final-prompt' : 'invariant-draft-with-pending-work',
  };
  for (const [name, value] of Object.entries(check)) {
    if (typeof value === 'boolean' && !value) throw new Error(item.id + ': readback failed: ' + name);
  }
  return check;
});
const readback = {

  routeFalpCallsMadeByThisScript: 0,

  outputCaseCount: reread.cases.length,
  readyForDeliveryAfterTextualReadback: checks.filter(item => item.stage === 'textually-reviewed-final-prompt').map(item => item.id),
  pending: checks.filter(item => item.stage === 'invariant-draft-with-pending-work').map(item => item.id),
  checks,
};
if (reread.variant !== 'motor-assisted' || reread.cases.length !== 8 || engine.routeFalpInvocationCount !== 8) {

}
