if (input.cases.length !== 8 || new Set(input.cases.map(item => item.id)).size !== 8) {
  throw new Error('The authorized input must contain eight distinct cases.');
}
const requests = input.cases.map(item => {
  const hostAssessment = assessments[item.id];
  if (!hostAssessment) throw new Error('Missing explicit host assessment: ' + item.id);
  const common = { context: structuredClone(item.context), explorationRequested: hostAssessment.explorationRequested };
  const request = hostAssessment.kind === 'ready'
    ? { kind: 'ready', ...common }
    : {
        kind: 'investigation',
        ...common,
        classification: hostAssessment.classification,
        state: compileAnchor(item.brief),
      };
  return { id: item.id, hostAssessment, request };
});

const startedAt = new Date().toISOString();
const results = [];
for (const [index, item] of requests.entries()) {
  const calledAt = new Date().toISOString();

  const result = routeFalp(item.request);
  const record = { id: item.id, invocation: index + 1, calledAt, result };

}
